📦 ÉTAPE 2 : Copier/Coller ces Commandes
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

# 1. Installer Node.js
curl -o- https://raw.githubusercontent.com/nvm-sh/nvm/v0.39.0/install.sh | bash
source ~/.bashrc
nvm install --lts
nvm use --lts

# 2. Installer PM2
npm install -g pm2

# 3. Installer PostgreSQL
sudo apt update
sudo apt install postgresql postgresql-contrib -y
sudo systemctl start postgresql
sudo systemctl enable postgresql

# 4. Créer la base de données
sudo -u postgres psql
# (Dans PostgreSQL, copiez/collez ces lignes :)
CREATE DATABASE garden_db;
CREATE USER garden_user WITH PASSWORD 'VotreMotDePasse123!';
GRANT ALL PRIVILEGES ON DATABASE garden_db TO garden_user;
ALTER DATABASE garden_db OWNER TO garden_user;
\q

# 5. Installer Nginx
sudo apt install nginx -y
sudo systemctl start nginx
sudo systemctl enable nginx

# 6. Cloner le projet
cd /var/www
git clone https://github.com/dryyss/Garden-Gold-Green.git garden-gold-green
cd garden-gold-green/garden-gold-green
npm install

🔧 ÉTAPE 3 : Configurer Nginx
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

sudo nano /etc/nginx/sites-available/garden-gold-green

# Collez cette configuration (remplacez votre domaine) :

server {
    listen 80;
    server_name VOTRE_DOMAINE.fr www.VOTRE_DOMAINE.fr;

    location / {
        proxy_pass http://localhost:3000;
        proxy_http_version 1.1;
        proxy_set_header Upgrade $http_upgrade;
        proxy_set_header Connection 'upgrade';
        proxy_set_header Host $host;
        proxy_cache_bypass $http_upgrade;
        proxy_set_header X-Real-IP $remote_addr;
        proxy_set_header X-Forwarded-For $proxy_add_x_forwarded_for;
        proxy_set_header X-Forwarded-Proto $scheme;
    }
}

# Enregistrer : Ctrl+X, puis Y, puis Enter

sudo ln -s /etc/nginx/sites-available/garden-gold-green /etc/nginx/sites-enabled/
sudo nginx -t
sudo systemctl restart nginx
