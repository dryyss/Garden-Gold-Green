# Commandes à Exécuter sur le Serveur IONOS (Copier/Coller)

# ========================================
# ÉTAPE 1 : Installation Node.js
# ========================================
curl -o- https://raw.githubusercontent.com/nvm-sh/nvm/v0.39.0/install.sh | bash
source ~/.bashrc
nvm install --lts
nvm use --lts
node --version

# ========================================
# ÉTAPE 2 : Installation PM2
# ========================================
npm install -g pm2

# ========================================
# ÉTAPE 3 : Installation PostgreSQL
# ========================================
sudo apt update
sudo apt install postgresql postgresql-contrib -y
sudo systemctl start postgresql
sudo systemctl enable postgresql

# Créer la base de données
sudo -u postgres psql

# Dans PostgreSQL, exécutez :
CREATE DATABASE garden_db;
CREATE USER garden_user WITH PASSWORD 'VotreMotDePasseFort123!';
GRANT ALL PRIVILEGES ON DATABASE garden_db TO garden_user;
ALTER DATABASE garden_db OWNER TO garden_user;
\q

# ========================================
# ÉTAPE 4 : Installation Nginx
# ========================================
sudo apt install nginx -y
sudo systemctl start nginx
sudo systemctl enable nginx

# Créer la configuration
sudo nano /etc/nginx/sites-available/garden-gold-green

# (Collez la configuration Nginx du guide)

# Activer le site
sudo ln -s /etc/nginx/sites-available/garden-gold-green /etc/nginx/sites-enabled/
sudo nginx -t
sudo systemctl restart nginx

# ========================================
# ÉTAPE 5 : Cloner le Projet
# ========================================
cd /var/www
git clone https://github.com/dryyss/Garden-Gold-Green.git garden-gold-green
cd garden-gold-green/garden-gold-green
npm install

# ========================================
# ÉTAPE 6 : Créer le fichier .env
# ========================================
nano .env

# (Copiez le contenu du .env du guide)

# ========================================
# ÉTAPE 7 : Build et Déployer
# ========================================
npx prisma generate
npx prisma migrate deploy
npm run build

# Lancer avec PM2
pm2 start npm --name "garden-gold-green" -- start
pm2 save
pm2 startup

# (Copiez et exécutez la commande affichée)

# ========================================
# ÉTAPE 8 : Configuration SSL
# ========================================
sudo apt install certbot python3-certbot-nginx -y
sudo certbot --nginx -d votredomaine.fr -d www.votredomaine.fr

# ========================================
# VÉRIFICATION
# ========================================
pm2 status
pm2 logs garden-gold-green
